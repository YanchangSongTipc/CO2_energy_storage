## 五、可复核方式

```powershell
# 统计引用覆盖率（孤儿文献 / 漏引）
python - <<'PY'
import re
t = open("ML_EnergyStorage_Optimization/paper_manuscript.md", encoding="utf-8").read()
body, refs = t.split("## References")
nums = [int(n) for n in re.findall(r"^\[(\d+)\]", refs, re.M)]
cited = set()
for grp in re.findall(r"\[([\d,\s\u2013-]+)\]", body):
    for tok in grp.split(","):
        tok = tok.strip().replace(" ", "")
        if re.match(r"^\d+[\u2013-]\d+$", tok):
            a, b = re.split(r"[\u2013-]", tok); cited.update(range(int(a), int(b)+1))
        elif tok.isdigit(): cited.add(int(tok))
print("orphans:", sorted(set(nums) - cited), "missing:", sorted(cited - set(nums)))
PY
```
